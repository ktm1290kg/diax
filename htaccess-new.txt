# DIAX — novi sajt (index.html). Stari WordPress je UGASEN:
# sve stare adrese + WP login/admin trajno preusmereni na pocetnu (301).
# Povratak na stari sajt: vratiti .htaccess.pre-diax preko ovog fajla.
DirectoryIndex index.html index.php

<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /
RewriteRule ^(wp-login\.php|xmlrpc\.php|wp-admin|adminer.*\.php|phpinfo\.php) https://diax.rs/ [R=301,L]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule . https://diax.rs/ [R=301,L]
</IfModule>
